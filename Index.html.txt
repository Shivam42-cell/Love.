    /* ----------------------------------------------------
       1. WEB AUDIO API CHIMES (NO EXTERNAL AUDIO REQUIRED)
       ---------------------------------------------------- */
    let audioCtx = null;
    let isAmbientPlaying = false;
    let ambientTimer = null;

    function getAudioContext() {
      if (!audioCtx) {
        audioCtx = new (window.AudioContext || window.webkitAudioContext)();
      }
      if (audioCtx.state === 'suspended') {
        audioCtx.resume();
      }
      return audioCtx;
    }

    function playTone(freq, type = 'sine', duration = 1.0, maxGain = 0.1) {
      try {
        const ctx = getAudioContext();
        const osc = ctx.createOscillator();
        const gain = ctx.createGain();

        osc.type = type;
        osc.frequency.setValueAtTime(freq, ctx.currentTime);

        gain.gain.setValueAtTime(0, ctx.currentTime);
        gain.gain.linearRampToValueAtTime(maxGain, ctx.currentTime + 0.04);
        gain.gain.exponentialRampToValueAtTime(0.0001, ctx.currentTime + duration);

        osc.connect(gain);
        gain.connect(ctx.destination);

        osc.start();
        osc.stop(ctx.currentTime + duration);
      } catch (e) {
        console.warn("Audio unavailable:", e);
      }
    }

    function playHarpChime() {
      const notes = [523.25, 659.25, 783.99, 1046.50]; // C5, E5, G5, C6
      notes.forEach((freq, idx) => {
        setTimeout(() => playTone(freq, 'sine', 1.5, 0.08), idx * 110);
      });
    }

      playTone(659.25, 'triangle', 0.25, 0.05);
    }

      });
    }

    const ambCanvas = document.getElementById('ambientCanvas');

    class LavenderPetal {
      constructor() {
        this.reset();
      }
      reset() {
        this.x = Math.random() * ambCanvas.width;
        this.y = Math.random() * -ambCanvas.height;
        this.size = Math.random() * 8 + 5;
        this.speedY = Math.random() * 0.8 + 0.4;
        this.speedX = Math.random() * 0.7 - 0.35;
        this.angle = Math.random() * 360;
        this.spin = Math.random() * 1.5 - 0.75;
        this.opacity = Math.random() * 0.5 + 0.3;
        const colors = ['rgba(197, 154, 238, ', 'rgba(221, 190, 247, ', 'rgba(171, 114, 227, '];
        this.color = colors[Math.floor(Math.random() * colors.length)];
      }
      update() {
        this.y += this.speedY;
        this.x += Math.sin(this.y * 0.006) + this.speedX;
        this.angle += this.spin;
        if (this.y > ambCanvas.height + 20 || this.x < -30 || this.x > ambCanvas.width + 30) {
          this.reset();
          this.y = -10;
        }
      }
      draw() {
        ambCtx.save();
        ambCtx.translate(this.x, this.y);
        ambCtx.rotate((this.angle * Math.PI) / 180);
        ambCtx.fillStyle = this.color + this.opacity + ')';
        ambCtx.beginPath();
        ambCtx.ellipse(0, 0, this.size, this.size / 2.2, Math.PI / 4, 0, Math.PI * 2);
        ambCtx.fill();
        ambCtx.restore();
      }

    function loopCatcherGame() {
      if (!catcherRunning) return;
      cCtx.clearRect(0, 0, catcherCanvas.width, catcherCanvas.height);

      // Draw Basket
      cCtx.fillStyle = '#914ed3';
      cCtx.beginPath();

          item.y >= catcherCanvas.height - basketHeight - 16 &&
          item.y <= catcherCanvas.height &&
          item.x + item.size >= basketX &&
          item.x <= basketX + basketWidth

    const wheelCanvas = document.getElementById('wheelCanvas');
    const wCtx = wheelCanvas.getContext('2d');
    const wheelSegments = [
      "100 Hugs On Demand",
      "Late Night Ice Cream",
      "Chef Mode: Favorite Meal",
      "1-Day Saying YES to Neha",
      "Long Hand-Holding Walk",
      "Post-Kaigo Coffee Date"
    ];
    const wheelColors = ['#f4ecfc', '#ebd9fa', '#ddbef7', '#c59aee', '#ab72e3', '#914ed3'];
    let currentWheelAngle = 0;
    let isSpinning = false;

    function drawWheel() {
      const size = wheelCanvas.width;
      const center = size / 2;
      const radius = center - 8;
      const sliceAngle = (2 * Math.PI) / wheelSegments.length;

      wCtx.clearRect(0, 0, size, size);

      wheelSegments.forEach((text, i) => {
        const start = i * sliceAngle;
        const end = start + sliceAngle;

        wCtx.beginPath();
        wCtx.moveTo(center, center);
        wCtx.arc(center, center, radius, start, end);
        wCtx.closePath();
        wCtx.fillStyle = wheelColors[i];
        wCtx.fill();
        wCtx.lineWidth = 2;
        wCtx.strokeStyle = '#ffffff';
        wCtx.stroke();

        // Draw Text
        wCtx.save();
        wCtx.translate(center, center);
        wCtx.rotate(start + sliceAngle / 2);
        wCtx.textAlign = 'right';
        wCtx.fillStyle = i >= 4 ? '#ffffff' : '#4a2373';
        wCtx.font = 'bold 11px sans-serif';
        wCtx.fillText(text, radius - 15, 4);
        wCtx.restore();
      });
    }

    function spinWheel() {
      if (isSpinning) return;
      isSpinning = true;
      playHarpChime();
      document.getElementById('wheelResult').classList.add('hidden');

      const extraRotations = 5 * 360;
      const randomOffset = Math.floor(Math.random() * 360);
      const targetDegree = extraRotations + randomOffset;

      wheelCanvas.style.transform = `rotate(${targetDegree}deg)`;

      setTimeout(() => {
        isSpinning = false;
        playWinFanfare();
        confetti({ particleCount: 70, spread: 60, origin: { y: 0.6 } });

        // Calculate winning slice (pointer is at top: 270 deg)
        const normalizedDegree = (360 - (targetDegree % 360) + 270) % 360;
        const sliceSize = 360 / wheelSegments.length;
        const winningIndex = Math.floor(normalizedDegree / sliceSize) % wheelSegments.length;

        document.getElementById('wheelPrizeText').innerText = wheelSegments[winningIndex];
        document.getElementById('wheelResult').classList.remove('hidden');
      }, 4000);
    }
